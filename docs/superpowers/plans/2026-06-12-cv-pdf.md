# CV Page with PDF Generation — Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Add a `/cv` page with dark sidebar layout, bilingual support, category-filtered project list, and client-side PDF export via pdfmake.

**Architecture:** Static Astro page reads `src/data/cv.ts` and the `projects` content collection at build time; data is injected into the page via `define:vars` for a client-side script that handles category filtering and pdfmake PDF generation. No server needed.

**Tech Stack:** Astro 4, TypeScript, pdfmake 0.2.x, CSS custom properties (existing design tokens)

import { getCollection } from 'astro:content';

const allProjects = await getCollection('projects');
const projects = allProjects
  .sort((a, b) => {
    const toMs = (d?: string) => d ? new Date(d + '-01').getTime() : 0;
    return toMs(b.data.startDate) - toMs(a.data.startDate);
  });

function formatYM(ym?: string | null): string {
  if (!ym) return '';
  const [year, month] = ym.split('-');
  const months = ['Jan','Feb','Mar','Apr','May','Jun','Jul','Aug','Sep','Oct','Nov','Dec'];
  return `${months[parseInt(month) - 1]} ${year}`;
}

function skillBar(level: number): string {
  const filled = Math.round(level / 20);
  return '█'.repeat(filled) + '░'.repeat(5 - filled);
}

const projectsForJs = projects.map(p => ({
  slug:         p.slug,
  title:        p.data.title,
  titlePt:      p.data.titlePt || p.data.title,
  description:  p.data.description,
  descriptionPt: p.data.descriptionPt || p.data.description,

  endDate:      p.data.endDate ?? null,
  position:     p.data.position ?? null,
  company:      p.data.company ?? null,
}));

const categories = [...new Set(projectsForJs.map(p => p.category))].sort();

<script define:vars={{ cvProjects: projectsForJs }}>
  // ── Helpers ──────────────────────────────────────────────────────────────────

  function fmtYM(ym) {
    if (!ym) return '';
    const [year, month] = ym.split('-');
    const months = ['Jan','Feb','Mar','Apr','May','Jun','Jul','Aug','Sep','Oct','Nov','Dec'];
    return `${months[parseInt(month) - 1]} ${year}`;
  }

  function getLang() {
    return document.documentElement.getAttribute('data-lang') || 'pt-BR';
  }

  function activeCategories() {
    return Array.from(document.querySelectorAll('.cv-chip.active')).map(el => el.dataset.cat);
  }

  // ── Project renderer ─────────────────────────────────────────────────────────

  function renderProjects() {
    const lang   = getLang();
    const active = activeCategories();
    const filtered = cvProjects.filter(p => active.includes(p.category));
    const container = document.getElementById('cv-projects');
    const counter   = document.getElementById('cv-projects-count');
    if (!container) return;

    if (filtered.length === 0) {
      container.innerHTML = '<p style="color:var(--text-muted);font-size:0.8rem;">No projects match the selected categories.</p>';
      if (counter) counter.textContent = '';
      return;
    }

    container.innerHTML = filtered.map(p => {
      const title = lang === 'pt-BR' ? p.titlePt : p.title;
      const desc  = lang === 'pt-BR' ? p.descriptionPt : p.description;
      const role  = p.position ? `${p.position}${p.company ? ' · ' + p.company : ''}` : (p.company || '');
      const dates = p.startDate
        ? `${fmtYM(p.startDate)} – ${p.endDate ? fmtYM(p.endDate) : (lang === 'pt-BR' ? 'atual' : 'present')}`
        : '';

      return `<div class="cv-project-item">
        <div class="cv-project-header">
          <span class="cv-project-title">${title}</span>
          ${dates ? `<span class="cv-project-dates">${dates}</span>` : ''}
        </div>
        ${role ? `<div class="cv-project-role">${role}</div>` : ''}
        <div class="cv-project-tags">${p.tech.map(t => `<span class="cv-project-tag">${t}</span>`).join('')}</div>
        <p class="cv-project-desc">${desc}</p>
      </div>`;
    }).join('');

    if (counter) {
      const total = cvProjects.length;
      counter.textContent = filtered.length < total
        ? `Showing ${filtered.length} of ${total} projects`
        : '';
    }
  }

  // ── Filter chips ─────────────────────────────────────────────────────────────

  document.getElementById('cv-filters')?.addEventListener('click', e => {
    const chip = (e.target).closest('.cv-chip');
    if (!chip) return;
    chip.classList.toggle('active');
    renderProjects();
  });

  // ── Language change ───────────────────────────────────────────────────────────

  document.documentElement.addEventListener('langchange', () => renderProjects());

  // ── PDF download ──────────────────────────────────────────────────────────────

  document.getElementById('cv-download')?.addEventListener('click', async () => {
    const btn = document.getElementById('cv-download');
    if (btn) btn.textContent = '⏳ Generating…';

    try {
      const [pdfMakeModule, pdfFontsModule] = await Promise.all([
        import('pdfmake/build/pdfmake'),
        import('pdfmake/build/vfs_fonts'),
      ]);
      const pdfMake = pdfMakeModule.default;
      pdfMake.vfs = pdfFontsModule.default.pdfMake.vfs;

      const lang    = getLang();
      const active  = activeCategories();
      const filtered = cvProjects.filter(p => active.includes(p.category));

      // Fetch photo as base64
      let photoData = null;
      try {
        const res    = await fetch('/img/fran.jpg');
        const blob   = await res.blob();
        photoData    = await new Promise(resolve => {
          const reader = new FileReader();
          reader.onloadend = () => resolve(reader.result);
          reader.readAsDataURL(blob);
        });
      } catch (_) { /* photo optional */ }

      const GREEN  = '#00ff87';
      const CYAN   = '#00d4ff';
      const AMBER  = '#ffb800';
      const DARK   = '#161b22';
      const TEXT   = '#e6edf3';
      const MUTED  = '#8b949e';
      const BG     = '#0d1117';
      const BORDER = '#30363d';

      function sectionLabel(text) {
        return { text, fontSize: 7, bold: true, color: AMBER, margin: [0, 10, 0, 4] };
      }

      function expItem(exp) {
        const pos   = lang === 'pt-BR' ? exp.position.pt : exp.position.en;
        const desc  = lang === 'pt-BR' ? exp.description.pt : exp.description.en;
        const start = fmtYM(exp.startDate);
        const end   = exp.endDate ? fmtYM(exp.endDate) : (lang === 'pt-BR' ? 'atual' : 'present');
        return {
          stack: [
            { columns: [
                { text: pos,  fontSize: 9, bold: true, color: TEXT,  width: '*' },
                { text: `${start} – ${end}`, fontSize: 7, color: MUTED, width: 'auto' },
            ]},
            { text: exp.company, fontSize: 8, color: CYAN, margin: [0, 1, 0, 2] },
            { text: desc, fontSize: 8, color: MUTED, lineHeight: 1.4 },
          ],
          margin: [0, 0, 0, 8],
          // left border simulation
        };
      }

      function projItem(p) {
        const title = lang === 'pt-BR' ? p.titlePt : p.title;
        const desc  = lang === 'pt-BR' ? p.descriptionPt : p.description;
        const role  = p.position ? `${p.position}${p.company ? ' · ' + p.company : ''}` : (p.company || '');
        const start = fmtYM(p.startDate);
        const end   = p.endDate ? fmtYM(p.endDate) : (lang === 'pt-BR' ? 'atual' : 'present');
        const dates = start ? `${start} – ${end}` : '';
        return {
          stack: [
            { columns: [
                { text: title, fontSize: 9, bold: true, color: TEXT,  width: '*' },
                { text: dates, fontSize: 7, color: MUTED, width: 'auto' },
            ]},
            ...(role ? [{ text: role, fontSize: 8, color: CYAN, margin: [0, 1, 0, 1] }] : []),
            { text: p.tech.join(' · '), fontSize: 7, color: CYAN, margin: [0, 0, 0, 2] },
            { text: desc, fontSize: 8, color: MUTED, lineHeight: 1.4 },
          ],
          margin: [0, 0, 0, 8],
        };
      }

      const sidebarStack = [
        ...(photoData ? [{ image: photoData, width: 80, alignment: 'center', margin: [0, 16, 0, 8] }] : [{ text: '', margin: [0, 16, 0, 0] }]),
        { text: 'FRANCIS PIRES', fontSize: 11, bold: true, color: TEXT, alignment: 'center', margin: [0, 0, 0, 3] },
        { text: lang === 'pt-BR' ? 'Engenheiro de Software' : 'Software Engineer', fontSize: 8, color: GREEN, alignment: 'center', margin: [0, 0, 0, 14] },
        sectionLabel('── CONTACT ──'),
        { text: `📧 ${window.__cvData.contact.email}`,   fontSize: 8, color: MUTED, margin: [0, 2, 0, 0] },
        { text: `🔗 ${window.__cvData.contact.github}`,  fontSize: 8, color: MUTED, margin: [0, 2, 0, 0] },
        { text: `🌐 ${window.__cvData.contact.site}`,    fontSize: 8, color: MUTED, margin: [0, 2, 0, 0] },
        { text: `📍 ${lang === 'pt-BR' ? window.__cvData.contact.city.pt : window.__cvData.contact.city.en}`, fontSize: 8, color: MUTED, margin: [0, 2, 0, 0] },
        sectionLabel('── SKILLS ──'),
        ...window.__cvData.skills.map(s => ({
          columns: [
            { text: s.name, fontSize: 7, color: MUTED, width: '*' },
            { text: '█'.repeat(Math.round(s.level / 20)) + '░'.repeat(5 - Math.round(s.level / 20)), fontSize: 7, color: GREEN, width: 'auto' },
          ],
          margin: [0, 2, 0, 0],
        })),
        sectionLabel('── EDUCATION ──'),
        ...window.__cvData.education.map(edu => ({
          stack: [
            { text: edu.institution, fontSize: 8, bold: true, color: CYAN },
            { text: lang === 'pt-BR' ? edu.degree.pt : edu.degree.en, fontSize: 7, color: TEXT, margin: [0, 1, 0, 1] },
            { text: `${edu.startYear}–${edu.endYear}`, fontSize: 7, color: MUTED },
          ],
          margin: [0, 0, 0, 6],
        })),
      ];

      const mainStack = [
        { text: lang === 'pt-BR' ? '$ whoami' : '$ whoami', fontSize: 8, color: GREEN, margin: [0, 16, 0, 4] },
        { text: lang === 'pt-BR' ? window.__cvData.summary.pt : window.__cvData.summary.en, fontSize: 8, color: MUTED, lineHeight: 1.5, margin: [0, 0, 0, 14] },
        { text: '$ cat experience.log', fontSize: 8, color: GREEN, margin: [0, 0, 0, 6] },
        ...window.__cvData.experience.map(expItem),
        { text: '$ ls projects/ | sort -r', fontSize: 8, color: CYAN, margin: [0, 6, 0, 6] },
        ...filtered.map(projItem),
      ];

      const docDef = {
        pageSize:    'A4',
        pageMargins: [0, 0, 0, 0],
        background:  { canvas: [{ type: 'rect', x: 0, y: 0, w: 595, h: 842, color: BG }] },
        content: [{
          columns: [
            { width: 180, stack: sidebarStack, fillColor: DARK, margin: [10, 0, 10, 0] },
            { width: '*',  stack: mainStack,   margin: [16, 0, 20, 0] },
          ],
        }],
        defaultStyle: { font: 'Roboto', fontSize: 9, color: TEXT },
      };

      pdfMake.createPdf(docDef).download('francis-pires-cv.pdf');
    } finally {
      const btn = document.getElementById('cv-download');
      if (btn) btn.textContent = '⬇ Download PDF';
    }
  });

  // ── Boot ──────────────────────────────────────────────────────────────────────

  // Expose cv static data for the PDF builder (set by inline script below)
  window.__cvData = window.__cvData || {};
  renderProjects();

- [ ] **Step 3: Add writeProject function**

Add this function before `async function main()`:

```js
async function writeProject(fields) {
  const {
    title, titlePt, description, descriptionPt,
    position, company, startDate, endDate,
    tech, github, url, featured, category,
  } = fields;

  const date    = new Date().toISOString().split('T')[0];
  const slug    = slugify(title, { lower: true, strict: true });
  const techArr = tech.split(',').map(t => t.trim()).filter(Boolean);

  const lines = ['---'];
  lines.push(`title: ${JSON.stringify(title)}`);
  if (titlePt)       lines.push(`titlePt: ${JSON.stringify(titlePt)}`);
  lines.push(`description: ${JSON.stringify(description)}`);
  if (descriptionPt) lines.push(`descriptionPt: ${JSON.stringify(descriptionPt)}`);
  lines.push(`tech: [${techArr.map(t => JSON.stringify(t)).join(', ')}]`);
  if (url)           lines.push(`url: ${JSON.stringify(url)}`);
  if (github)        lines.push(`github: ${JSON.stringify(github)}`);
  lines.push(`featured: ${featured}`);
  lines.push(`order: 99`);
  if (category)      lines.push(`category: ${category}`);
  if (position)      lines.push(`position: ${JSON.stringify(position)}`);
  if (company)       lines.push(`company: ${JSON.stringify(company)}`);
  lines.push(`startDate: "${startDate}"`);
  if (endDate)       lines.push(`endDate: "${endDate}"`);
  lines.push('---', '');

  const projectsDir = join(ROOT, 'src/content/projects');
  mkdirSync(projectsDir, { recursive: true });
  const filename = `${date}-${slug}.md`;
  const filepath = join(projectsDir, filename);
  writeFileSync(filepath, lines.join('\n') + '\n');

  console.log('\n' + chalk.green('✓ Project created!\n'));
  console.log('  ' + chalk.cyan(`src/content/projects/${filename}`));
  console.log('\n' + chalk.dim('Next steps:'));
  console.log('  1. ' + chalk.white('Add content to the file (body after the frontmatter)'));
  console.log('  2. ' + chalk.white('npm run dev') + chalk.dim('  →  http://localhost:4321/projects'));
  console.log('  3. ' + chalk.white('git add src/content/projects/ && git commit -m "project: ' + title + '"'));
}
```
